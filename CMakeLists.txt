function(copy_thirdparty_licenses TARGET_NAME BUNDLE_DEST)
    # Automatically locate all license files in thirdparty/
    file(GLOB_RECURSE LICENSE_FILES
        CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/*LICENSE*"
        "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/*LICENCE*"
        "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/*OFL*"
        "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/*COPYING*"
        "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/*GUST*"
    )

    set(COPY_COMMANDS "")

    foreach(SRC_PATH IN LISTS LICENSE_FILES)
        # Get path relative to thirdparty/ directory
        file(RELATIVE_PATH REL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty" "${SRC_PATH}")

        # Convert slashes to dashes so "implot/LICENSE" becomes "implot-LICENSE"
        string(REPLACE "/" "-" DEST_FILENAME "${REL_PATH}")
        string(REPLACE "\\" "-" DEST_FILENAME "${DEST_FILENAME}")

        list(APPEND COPY_COMMANDS
            COMMAND ${CMAKE_COMMAND} -E copy
                "${SRC_PATH}"
                "${BUNDLE_DEST}/licenses/${DEST_FILENAME}"
        )
    endforeach()

    if(COPY_COMMANDS)
        add_custom_command(
            TARGET ${TARGET_NAME} POST_BUILD
            ${COPY_COMMANDS}
            COMMENT "Copying all thirdparty licenses to bundle..."
        )
    endif()
endfunction()

copy_thirdparty_licenses(${PROJECT_NAME} "${BUNDLE_DIR}")
